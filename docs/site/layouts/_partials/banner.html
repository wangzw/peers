{{- /* Mirrors the Hextra v0.13.0 banner partial. The version banner replaces the theme banner rather than stacking on it: class older or dev renders the version banner here and the theme's params.banner banner does not render at all, with no close button so it cannot be dismissed. Class latest keeps the theme's params.banner banner unchanged, close button included. A build that sets params.banner for any class other than latest fails, so the two banners are never both present. Because the theme emits its banner script only for params.banner, the version banner measures itself and sets --hextra-banner-height and data-hextra-banner, which the navbar and the mobile sidebar read. */ -}} {{- $version := partial "version/context.html" . -}} {{- $versionBanner := or (eq $version.class "older") (eq $version.class "dev") -}} {{- if and site.Params.banner (ne $version.class "latest") -}} {{- errorf "banner.html: params.banner is set for class %s; the site banner is supported only for class latest, where the version banner does not render" $version.class -}} {{- end -}} {{- if $versionBanner }} {{- $latestURL := "" -}} {{- $latestName := $version.latestName -}} {{- range $version.versions }} {{- if eq .class "latest" }} {{- $latestName = .name -}} {{- $latestURL = partial "version/in.html" (dict "base" .url "page" $) -}} {{- end }} {{- end }} {{- if not $latestURL }} {{- $latestURL = partial "version/in.html" (dict "base" $version.latestBaseURL "page" $) -}} {{- end }} {{- $latestLink := printf `%s` $latestURL $latestName | safeHTML -}} {{- $key := cond (eq $version.class "older") "versionBannerOlder" "versionBannerDev" -}}

{{- T $key (dict "Version" $version.name "Latest" $latestLink) | safeHTML -}}

{{- end }} {{- if and (not $versionBanner) site.Params.banner }}

{{- end -}}
