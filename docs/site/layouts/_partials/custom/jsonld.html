{{- /* JSON-LD for the site, generated from the page rather than written into the Markdown. The home page gets a SoftwareApplication naming the release the build documents and linking to it, a page marked `faq: true` gets a FAQPage built from its own headings and answers, and every docs page gets a BreadcrumbList. The FAQ page is also a docs page, so it gets both. The SoftwareApplication url and the BreadcrumbList items are the page's canonical URL. In class latest-copy that is the same path under canonicalBaseURL, so structured data points at the root copy, not at the /vX.Y.Z/ copy. */ -}} {{- $version := partial "version/context.html" . -}} {{- $lang := .Site.Language.Locale | default .Site.Language.Lang -}} {{- $canonical := .Store.Get "canonical" | default .Permalink -}} {{- if .IsHome -}} {{- $downloadURL := printf "https://github.com/wangzw/peers/releases/tag/v%s" $version.release -}} {{- if eq $version.class "dev" -}} {{- $downloadURL = printf "https://github.com/wangzw/peers/releases/tag/%s" $version.latestName -}} {{- end -}} {{- $app := dict "@context" "https://schema.org" "@type" "SoftwareApplication" "name" "peers" "description" .Site.Params.description "url" $canonical "applicationCategory" "DeveloperApplication" "operatingSystem" "Linux, macOS" "license" "https://www.apache.org/licenses/LICENSE-2.0" "downloadUrl" $downloadURL "offers" (dict "@type" "Offer" "price" "0" "priceCurrency" "USD") "sameAs" "https://github.com/wangzw/peers" "inLanguage" $lang "author" (dict "@type" "Person" "@id" "https://zhanwei.wang/#person" "name" "wangzw" "url" "https://zhanwei.wang/en/" "sameAs" (slice "https://github.com/wangzw")) -}} {{- if ne $version.class "dev" -}} {{- $app = merge $app (dict "softwareVersion" $version.release) -}} {{- end -}} {{- if not .Lastmod.IsZero -}} {{- $app = merge $app (dict "dateModified" (.Lastmod.Format "2006-01-02T15:04:05-07:00")) -}} {{- end -}} {{- end -}} {{- if eq .Section "docs" -}} {{- if .Params.faq -}} {{- /* Each level-two heading is a question; its answer is the plain text between that heading and the next one. */ -}} {{- $questions := slice -}} {{- $parts := split .Content "]*>" "" $part -}} {{- $halves := split $stripped "" -}} {{- $question := index $halves 0 | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- $answer := "" -}} {{- if gt (len $halves) 1 -}} {{- $answer = index $halves 1 | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- end -}} {{- if and $question $answer -}} {{- $questions = $questions | append (dict "@type" "Question" "name" $question "acceptedAnswer" (dict "@type" "Answer" "text" $answer)) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $questions -}} {{- $faq := dict "@context" "https://schema.org" "@type" "FAQPage" "inLanguage" $lang "mainEntity" $questions -}} {{- end -}} {{- end -}} {{- /* Breadcrumb trail from the home page down to this page. */ -}} {{- $crumbs := slice -}} {{- $ancestors := .Ancestors.Reverse -}} {{- range $i, $p := $ancestors -}} {{- $name := $p.LinkTitle | default $p.Title -}} {{- if $p.IsHome -}} {{- $name = site.Title -}} {{- end -}} {{- $item := $p.Permalink -}} {{- if eq $version.class "latest-copy" -}} {{- $item = partial "version/in.html" (dict "base" $version.canonicalBaseURL "page" $p) -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add $i 1) "name" $name "item" $item) -}} {{- end -}} {{- $currentItem := .Permalink -}} {{- if eq $version.class "latest-copy" -}} {{- $currentItem = $canonical -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "@type" "ListItem" "position" (add (len $ancestors) 1) "name" (.LinkTitle | default .Title) "item" $currentItem) -}} {{- $breadcrumb := dict "@context" "https://schema.org" "@type" "BreadcrumbList" "itemListElement" $crumbs -}} {{- end -}}
